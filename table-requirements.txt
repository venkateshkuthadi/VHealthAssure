----------- Users Table ---------------

CREATE TABLE users (
    UserId CHAR(36) NOT NULL,
    FullName VARCHAR(150) NOT NULL,
    Email VARCHAR(255) NOT NULL,
    Mobile VARCHAR(20) NOT NULL,
    Password VARCHAR(255) NOT NULL,
    GovIdType ENUM('AADHAAR','PAN','DL','PASSPORT','VOTER') NOT NULL,
    GovIdNumber VARCHAR(64) NOT NULL,
    DateOfBirth DATE DEFAULT NULL,
    Gender ENUM('MALE','FEMALE','OTHER') DEFAULT NULL,
    AddressLine1 VARCHAR(255) DEFAULT NULL,
    AddressLine2 VARCHAR(255) DEFAULT NULL,
    City VARCHAR(100) DEFAULT NULL,
    State VARCHAR(100) DEFAULT NULL,
    PostalCode VARCHAR(20) DEFAULT NULL,
    Country VARCHAR(100) DEFAULT NULL,
    NomineeName VARCHAR(150) DEFAULT NULL,
    NomineeRelation VARCHAR(50) DEFAULT NULL,
    NomineeDob DATE DEFAULT NULL,
    NomineeShare DECIMAL(5,2) DEFAULT 100.00,
    EmploymentStatus ENUM('SALARIED','SELF_EMPLOYED','UNEMPLOYED','RETIRED') DEFAULT NULL,
    AnnualIncome DECIMAL(14,2) DEFAULT 0.00,
    Occupation VARCHAR(100) DEFAULT NULL,
    Company VARCHAR(150) DEFAULT NULL,
    EmailVerified TINYINT(1) DEFAULT 0,
    MobileVerified TINYINT(1) DEFAULT 0,
    KycStatus ENUM('NOT VERIFIED','PENDING','VERIFIED','REJECTED') DEFAULT 'NOT VERIFIED',
    Role ENUM('USER','AGENT','ADMIN') DEFAULT 'USER',
    Status ENUM('ACTIVE','INACTIVE','SUSPENDED') DEFAULT 'ACTIVE',
    CreatedAt DATETIME DEFAULT CURRENT_TIMESTAMP,
    UpdatedAt DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (UserId),
    UNIQUE KEY uk_users_email (Email),
    UNIQUE KEY uk_users_mobile (Mobile),
    UNIQUE KEY uk_users_govidnumber (GovIdNumber),
    KEY idx_users_govidtype (GovIdType)
);
